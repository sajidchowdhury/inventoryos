    23→function calcTitleLayout(title, maxWidthTwips, preferredPt = 40, minPt = 24) {
    24→  const charWidth = (pt) => pt * 11;
    25→  const charsPerLine = (pt) => Math.floor(maxWidthTwips / charWidth(pt));
    26→  let titlePt = preferredPt, lines;
    27→  while (titlePt >= minPt) {
    28→    const cpl = charsPerLine(titlePt);
    29→    if (cpl < 2) { titlePt -= 2; continue; }
    30→    lines = splitTitleLines(title, cpl);
    31→    if (lines.length <= 3) break;
    32→    titlePt -= 2;
    33→  }
    34→  if (!lines || lines.length > 3) {
    35→    lines = splitTitleLines(title, charsPerLine(minPt));
    36→    titlePt = minPt;
    37→  }
    38→  return { titlePt, titleLines: lines };
    39→}
    40→
    41→function splitTitleLines(title, charsPerLine) {
    42→  if (title.length <= charsPerLine) return [title];
    43→  const breakAfter = new Set([' ', '-', '/', '(', ')', ':', ',']);
    44→  const lines = []; let remaining = title;
    45→  while (remaining.length > charsPerLine) {
    46→    let breakAt = -1;
    47→    for (let i = charsPerLine; i >= Math.floor(charsPerLine * 0.6); i--) {
    48→      if (i < remaining.length && breakAfter.has(remaining[i - 1])) { breakAt = i; break; }
    49→    }
    50→    if (breakAt === -1) {
    51→      const limit = Math.min(remaining.length, Math.ceil(charsPerLine * 1.3));
    52→      for (let i = charsPerLine + 1; i < limit; i++) {
    53→        if (breakAfter.has(remaining[i - 1])) { breakAt = i; break; }
    54→      }
    55→    }
    56→    if (breakAt === -1) breakAt = charsPerLine;
    57→    lines.push(remaining.slice(0, breakAt).trim());
    58→    remaining = remaining.slice(breakAt).trim();
    59→  }
    60→  if (remaining) lines.push(remaining);
    61→  if (lines.length > 1 && lines[lines.length - 1].length <= 3) {
    62→    const last = lines.pop(); lines[lines.length - 1] += last;
    63→  }
    64→  return lines;
    65→}

    67→function calcCoverSpacing(params) {
    68→  const { titleLineCount = 1, titlePt = 36, hasSubtitle = false, metaLineCount = 0, fixedHeight = 400 } = params;
    69→  const SAFETY = 1200;
    70→  const usableHeight = 16838 - 0 - 0 - SAFETY;

    72→  const subtitleHeight = hasSubtitle ? (12 * 23 + 600) : 0;
    73→  const metaHeight = metaLineCount * (10 * 23 + 100);
    74→  const contentHeight = titleHeight + subtitleHeight + metaHeight + fixedHeight + 900;
    75→  const remainingSpace = Math.max(usableHeight - contentHeight, 400);
    76→  const FOOTER_MIN = 800;
    77→  const rawBottom = Math.floor(remainingSpace * 0.45);
    78→  const bottomSpacing = Math.max(rawBottom, FOOTER_MIN);
    79→  const topSpacing = Math.max(Math.floor(remainingSpace * 0.45) - Math.max(0, FOOTER_MIN - rawBottom), 400);
    80→  return { topSpacing, midSpacing: 0, bottomSpacing };
