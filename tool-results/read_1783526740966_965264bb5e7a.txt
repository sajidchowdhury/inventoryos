   724→// ── OTP Step ──
   725→function OtpStep() {
   726→  const { phone, setStep, setUserId, setUserName, setBusinesses, setIsLoading, setError, isLoading, error, trustDevice, setPhoneToken, setDeviceToken } = useAuthStore();
   727→  const [otp, setOtp] = useState(["", "", "", ""]);
   728→  const [resendCooldown, setResendCooldown] = useState(0);
   729→
   730→  // Cooldown timer for resend
   731→  useState(() => {
   732→    if (resendCooldown <= 0) return;
   733→    const timer = setTimeout(() => setResendCooldown(resendCooldown - 1), 1000);
   734→    return () => clearTimeout(timer);
   735→  });
   736→
   737→  const handleVerify = async (code?: string) => {
   738→    const otpCode = code || otp.join("");
   739→    if (otpCode.length < 4) return;
   740→
   741→    setIsLoading(true);
   742→    setError(null);
   743→    try {
   744→      const res = await fetch("/api/auth/verify-otp", {
   745→        method: "POST",
   746→        headers: { "Content-Type": "application/json" },
   747→        body: JSON.stringify({ phone, otp: otpCode, trustDevice }),
   748→      });
   749→      const data = await res.json();
   750→      if (!res.ok) throw new Error(data.error);
   751→
   752→      setUserId(data.user.id);
   753→      setUserName(data.user.name);
   754→      setBusinesses(data.businesses);
   755→      setPhoneToken(data.phoneToken ?? null);
   756→      if (data.deviceToken) setDeviceToken(data.deviceToken);
   757→
   758→      // Route based on whether they have businesses
   759→      if (data.businesses.length > 0) {
   760→        setStep("discovery");
   761→      } else {
   762→        setStep("add-business");
   763→      }
   764→    } catch (err: unknown) {
   765→      setError(err instanceof Error ? err.message : "Verification failed");
   766→    } finally {
   767→      setIsLoading(false);
   768→    }
   769→  };
   770→
   771→  const handleResend = async () => {
   772→    if (resendCooldown > 0) return;
   773→    setIsLoading(true);
   774→    setError(null);
   775→    try {
   776→      const res = await fetch("/api/auth/send-otp", {
   777→        method: "POST",
   778→        headers: { "Content-Type": "application/json" },
   779→        body: JSON.stringify({ phone }),
   780→      });
   781→      const data = await res.json();
   782→      if (!res.ok) throw new Error(data.error);
   783→      setResendCooldown(60);
   784→      setOtp(["", "", "", ""]);
   785→    } catch (err: unknown) {
   786→      setError(err instanceof Error ? err.message : "Failed to resend OTP");
   787→    } finally {
   788→      setIsLoading(false);
   789→    }
   790→  };
   791→
   792→  const handleChange = (index: number, value: string) => {
   793→    if (!/^\d*$/.test(value)) return;
   794→    const newOtp = [...otp];
   795→    newOtp[index] = value.slice(-1);
   796→    setOtp(newOtp);
   797→
   798→    // Auto-focus next input
   799→    if (value && index < 3) {
   800→      const nextInput = document.getElementById(`otp-${index + 1}`);
   801→      nextInput?.focus();
   802→    }
   803→
   804→    // Auto-verify when all 4 digits entered
   805→    if (newOtp.every((d) => d !== "") && newOtp.join("").length === 4) {
   806→      handleVerify(newOtp.join(""));
   807→    }
   808→  };
   809→
   810→  const handleKeyDown = (index: number, e: React.KeyboardEvent) => {
   811→    if (e.key === "Backspace" && !otp[index] && index > 0) {
   812→      const prevInput = document.getElementById(`otp-${index - 1}`);
   813→      prevInput?.focus();
   814→    }
   815→  };
   816→
   817→  return (
   818→    <motion.div {...slideIn} className="space-y-6">
   819→      <div className="text-center space-y-3">
   820→        <div className="inline-flex items-center justify-center h-16 w-16 rounded-2xl bg-gradient-to-br from-emerald-500 to-emerald-600 shadow-lg shadow-emerald-500/30 animate-pulse-soft">
   821→          <ShieldCheck className="h-8 w-8 text-white" />
   822→        </div>
   823→        <h2 className="text-2xl font-bold">Verify your number</h2>
   824→        <p className="text-muted-foreground text-sm">
   825→          Enter the 4-digit code sent to{" "}
   826→          <span className="font-semibold text-foreground">+880{phone}</span>
   827→        </p>
   828→      </div>
   829→
   830→      <div className="flex justify-center gap-3">
   831→        {otp.map((digit, i) => (
   832→          <Input
   833→            key={i}
   834→            id={`otp-${i}`}
   835→            type="text"
   836→            inputMode="numeric"
   837→            maxLength={1}
   838→            value={digit}
   839→            onChange={(e) => handleChange(i, e.target.value)}
   840→            onKeyDown={(e) => handleKeyDown(i, e)}
   841→            className="w-14 h-14 text-center text-2xl font-bold p-0 rounded-2xl border-emerald-200/60 focus-visible:ring-emerald-500/30 focus-visible:border-emerald-500 shadow-sm"
   842→            autoFocus={i === 0}
   843→          />
   844→        ))}
   845→      </div>
   846→
   847→      <div className="bg-emerald-50 border border-emerald-200 rounded-xl px-3 py-2 text-center">
   848→        <p className="text-xs text-emerald-800">
   849→          Demo OTP: <span className="font-mono font-bold text-emerald-700">9999</span>
   850→        </p>
   851→      </div>
   852→
   853→      {error && (
   854→        <p className="text-sm text-destructive text-center">{error}</p>
   855→      )}
   856→
   857→      <Button
   858→        size="lg"
   859→        className="w-full h-12 gap-2 rounded-2xl bg-gradient-to-r from-emerald-500 to-emerald-600 hover:from-emerald-600 hover:to-emerald-700 shadow-lg shadow-emerald-500/30 border-0 text-white"
   860→        onClick={() => handleVerify()}
   861→        disabled={otp.some((d) => !d) || isLoading}
   862→      >
   863→        {isLoading ? "Verifying..." : "Verify"}
   864→        {!isLoading && <CheckCircle2 className="h-4 w-4" />}
   865→      </Button>
   866→
   867→      {/* Resend OTP */}
   868→      <div className="text-center">
   869→        {resendCooldown > 0 ? (
   870→          <p className="text-sm text-muted-foreground">
   871→            Resend code in <span className="font-semibold">{resendCooldown}s</span>
   872→          </p>
   873→        ) : (
   874→          <Button variant="ghost" size="sm" className="gap-1.5" onClick={handleResend} disabled={isLoading}>
   875→            <RefreshCw className="h-3.5 w-3.5" /> Resend Code
   876→          </Button>
   877→        )}
   878→      </div>
   879→
   880→      <Button variant="ghost" className="w-full gap-2" onClick={() => setStep("phone")}>
   881→        <ArrowLeft className="h-4 w-4" /> Change phone number
   882→      </Button>
   883→    </motion.div>
   884→  );
   885→}

  1528→// ── Main Page ──

  1533→  // On mount, check if we have a persisted session and redirect to dashboard
  1534→  // This setState-in-effect is intentional: we need to hydrate from localStorage
  1535→  // after the first render to avoid SSR/client mismatch
  1536→  const [ready, setReady] = useState(false);
  1537→
  1538→  useEffect(() => {
  1539→    let cancelled = false;
  1540→
  1541→    const hydrate = async () => {

  1554→          const res = await fetch("/api/auth/trusted-device", {
