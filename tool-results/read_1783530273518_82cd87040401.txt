    43→// ── Trusted Devices (skip-OTP for returning owners) ──
    44→// A long-lived, revocable token that proves a device has previously verified
    45→// this phone via OTP. Lets an owner return to their business list without
    46→// re-entering an OTP until the token expires.
    47→model TrustedDevice {
    48→  id         String   @id @default(cuid())
    49→  userId     String
    50→  token      String   @unique
    51→  deviceInfo String?
    52→  expiresAt  DateTime
    53→  createdAt  DateTime @default(now())
    54→  lastUsedAt DateTime @default(now())
    55→
    56→  user User @relation(fields: [userId], references: [id], onDelete: Cascade)
    57→
    58→  @@index([userId])
    59→  @@index([token])
    60→  @@index([expiresAt])
    61→}

    63→// ── Phone Auth Tokens (short-lived OTP proof) ──
    64→// Issued by verify-otp / trusted-device. Proves the holder recently verified
    65→// ownership of a phone number. Required by owner-login so a raw userId from the
    66→// client can never be used to enter someone else's shop without OTP.
    67→model PhoneAuthToken {

    74→  user User @relation(fields: [userId], references: [id], onDelete: Cascade)
    75→
    76→  @@index([token])

  1064→// Recurring seasonal periods that shift baseline demand for months at a time.

  1069→// Pre-seeded with 4 Bangladesh seasons:
  1070→//   - Winter (Dec-Feb): respiratory meds spike 1.5x
  1071→//   - Summer (Mar-May): ORS, antacids, heatstroke meds 1.3x (overlaps Ramadan/Eid)
